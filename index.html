async function loadPosts() { try { const res = await fetch('https://api.github.com/repos/vinhhello13/hackchua/contents/_posts'); const files = await res.json(); const container = document.getElementById('cardContainer'); if (!container) return; container.innerHTML = ''; if (!Array.isArray(files) || files.length === 0) { container.innerHTML = '
Chưa có bài viết nào được đăng. Hãy vào /admin để đăng bài!
'; return; } for (const file of files) { if (file.name.endsWith('.md')) { const contentRes = await fetch(file.download_url); const rawText = await contentRes.text(); const parts = rawText.split('---'); let title = "Script VIP"; let externalHTML = ""; // Phần hiện ngoài trang chủ let internalHTML = ""; // Phần hiện trong popup chi tiết if (parts.length >= 3) { const yamlHeader = parts[1]; const fullMarkdown = parts.slice(2).join('---').trim(); const tMatch = yamlHeader.match(/title:\s*"?(.*?)"?$/m); if (tMatch) title = tMatch[1]; // Tách bằng chuẩn phong cách Blogger if (fullMarkdown.includes('')) { const splitParts = fullMarkdown.split(''); externalHTML = splitParts[0].trim(); internalHTML = splitParts[1].trim(); } else { externalHTML = fullMarkdown; internalHTML = fullMarkdown; } } else { externalHTML = rawText; internalHTML = rawText; } // Tạo thẻ Card ngoài trang chủ (Chỉ hiện phần trên ) const postCard = document.createElement('div'); postCard.className = 'glass-card rounded-2xl p-6 flex flex-col justify-between relative overflow-hidden group card-item border border-slate-800 col-span-1 md:col-span-2'; postCard.innerHTML = `
WORKING 100%
${externalHTML}
`; // KHI BẤM "XEM SCRIPT", BUNG TOÀN BỘ KHÔNG GIAN CHỨC NĂNG BÊN DƯỚI VÀO POPUP postCard.querySelector('.btn-xem').addEventListener('click', () => { openModal(title, `
${internalHTML}
`); // Kích hoạt lại các đoạn mã script chức năng (check trạng thái, số dư) bên trong popup setTimeout(() => { const scripts = document.getElementById('generalModal').querySelectorAll('script'); scripts.forEach(oldScript => { const newScript = document.createElement('script'); newScript.text = oldScript.text; oldScript.parentNode.replaceChild(newScript, oldScript); }); }, 200); }); postCard.querySelector('.btn-mua').addEventListener('click', () => { buyProduct(title); }); container.appendChild(postCard); } } } catch (err) { console.error('Lỗi tải bài viết:', err); const container = document.getElementById('cardContainer'); if (container) { container.innerHTML = '
Không thể tải danh sách bài viết từ GitHub.
'; } } } loadPosts();
