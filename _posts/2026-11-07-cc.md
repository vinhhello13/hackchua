<script>
  const GIA_SCRIPT_VIP = 1000;
  let isPageLocked = false;

  function muaBangSoDu() {
    if (isPageLocked) {
      alert("⚠️ Bài viết đang trong thời gian BẢO TRÌ / CẬP NHẬT, không thể mua lúc này!");
      return;
    }

    let loggedUser = localStorage.getItem("logged_user");
    if (!loggedUser) {
      alert("⚠️ Vui lòng đăng nhập trước khi mua hàng!");
      return;
    }

    let currentBalance = parseInt(localStorage.getItem("logged_balance")) || 0;
    if (currentBalance < GIA_SCRIPT_VIP) {
      alert("❌ Số dư ví không đủ 1.000đ để mua Script này!");
      return;
    }

    let confirmMua = confirm("🛒 Xác nhận mua Script này với giá " + GIA_SCRIPT_VIP.toLocaleString("vi-VN") + "đ?");
    if (!confirmMua) return;

    let buyBtn = document.getElementById("buyBtn");
    buyBtn.innerText = "⏳ Đang xử lý...";
    buyBtn.disabled = true;

    fetch(SCRIPT_URL, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify({ 
        action: "deductBalance", 
        username: loggedUser, 
        amount: GIA_SCRIPT_VIP,
        note: "Mua script (ID: " + MY_POST_ID + ")"
      })
    })
    .then(res => res.json())
    .then(data => {
      buyBtn.innerText = "💎 Mua Ngay (Trừ 1.000đ Số Dư Ví)";
      buyBtn.disabled = false;

      if (data.status === "success") {
        localStorage.setItem("logged_balance", data.balance);
        const balSpan = document.getElementById("userBalanceText");
        if (balSpan) balSpan.innerText = parseInt(data.balance).toLocaleString("vi-VN") + "đ";
        
        const globalBal = document.getElementById("globalUserBalance");
        if (globalBal) globalBal.innerText = parseInt(data.balance).toLocaleString("vi-VN") + "đ";

        alert("✅ Thanh toán thành công! Giao dịch đã được ghi vào lịch sử.");
        document.getElementById("scriptText").value = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/napun87/stealaneggv2/refs/heads/main/codiumhub.lua"))()';
        document.getElementById("scriptVipOutput").style.display = "block";
      } else {
        alert("❌ Lỗi: " + (data.message || "Không thể trừ tiền!"));
      }
    })
    .catch(err => {
      buyBtn.innerText = "💎 Mua Ngay (Trừ 1.000đ Số Dư Ví)";
      buyBtn.disabled = false;
      alert("❌ Lỗi kết nối đến máy chủ!");
    });
  }

  function copyScript() {
    let copyText = document.getElementById("scriptText");
    copyText.select();
    navigator.clipboard.writeText(copyText.value);
    alert("📋 Đã copy Script thành công!");
  }

  function toggleBaoLoi() {
    let box = document.getElementById("formBaoLoiBox");
    box.style.display = (box.style.display === "none") ? "block" : "none";
  }

  function guiBaoLoi() {
    let loggedUser = localStorage.getItem("logged_user") || "Chưa đăng nhập";
    let tieuDe = document.getElementById("loiTieuDe").value;
    let noiDungChiTiet = document.getElementById("loiNoiDung").value.trim();
    let fullMessage = "[" + tieuDe + "] [ID: " + MY_POST_ID + "]" + (noiDungChiTiet ? " " + noiDungChiTiet : "");

    isPageLocked = true;
    let tag = document.getElementById("statusTag");
    if (tag) {
      tag.innerText = "🔴 BẢO TRÌ / ĐANG CẬP NHẬT";
      tag.style.color = "#fb7185";
    }

    const freeBox = document.getElementById("khoangFreeBox");
    const vipBox = document.getElementById("khoangVipBox");
    if (freeBox) freeBox.style.display = "none";
    if (vipBox) vipBox.style.display = "none";

    fetch(SCRIPT_URL, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify({ 
        action: "reportAndLock", 
        postId: MY_POST_ID, 
        username: loggedUser, 
        message: fullMessage 
      })
    })
    .then(res => res.json())
    .catch(err => console.log(err));

    alert("✅ Đã gửi báo lỗi! Trạng thái bài viết đã tự động chuyển sang BẢO TRÌ.");
    document.getElementById("loiNoiDung").value = "";
    document.getElementById("formBaoLoiBox").style.display = "none";
  }
</script>
