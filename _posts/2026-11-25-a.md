<script>
  (function () {
    const GIA_SCRIPT_VIP = 1000;
    const API_URL = typeof SCRIPT_URL !== "undefined" ? SCRIPT_URL : "";
    const POST_ID = typeof MY_POST_ID !== "undefined" && MY_POST_ID ? String(MY_POST_ID) : "steal-an-egg-keyless";
    let isPageLocked = false;

    function formatVND(amount) {
      const num = Number(amount);
      return (Number.isFinite(num) ? num : 0).toLocaleString("vi-VN") + "đ";
    }

    function syncBalanceUI(balance) {
      const safeBalance = Math.max(0, parseInt(balance, 10) || 0);
      localStorage.setItem("logged_balance", String(safeBalance));

      const balSpan = document.getElementById("userBalanceText");
      if (balSpan) balSpan.innerText = formatVND(safeBalance);

      const globalBal = document.getElementById("globalUserBalance");
      if (globalBal) globalBal.innerText = formatVND(safeBalance);
    }

    function setMaintenanceState(locked, message) {
      isPageLocked = Boolean(locked);
      const tag = document.getElementById("statusTag");
      const freeBox = document.getElementById("khoangFreeBox");
      const vipBox = document.getElementById("khoangVipBox");

      if (isPageLocked) {
        if (tag) {
          tag.innerText = message || "🔴 BẢO TRÌ / ĐANG CẬP NHẬT";
          tag.style.color = "#fb7185";
        }
        if (freeBox) freeBox.style.display = "none";
        if (vipBox) vipBox.style.display = "none";
      } else {
        if (tag) {
          tag.innerText = "🟢 Hoạt động";
          tag.style.color = "#34d399";
        }
        if (freeBox) freeBox.style.display = "block";
        if (vipBox) vipBox.style.display = "block";
      }
    }

    const idEl = document.getElementById("displayPostId");
    if (idEl) idEl.innerText = POST_ID;
    syncBalanceUI(localStorage.getItem("logged_balance") || 0);

    window.handleOpenNap = function () {
      if (typeof openNapModal === "function") {
        openNapModal("MOMO");
      } else {
        alert("⚠️ Chức năng nạp tiền chưa được kết nối.");
      }
    };

    window.muaBangSoDu = async function () {
      if (isPageLocked) {
        alert("⚠️ Bài viết đang trong thời gian BẢO TRÌ / CẬP NHẬT, không thể mua lúc này!");
        return;
      }

      if (!API_URL) {
        alert("❌ Thiếu cấu hình máy chủ (SCRIPT_URL).");
        return;
      }

      const loggedUser = localStorage.getItem("logged_user");
      if (!loggedUser) {
        alert("⚠️ Vui lòng đăng nhập trước khi mua hàng!");
        return;
      }

      const currentBalance = parseInt(localStorage.getItem("logged_balance"), 10) || 0;
      if (currentBalance < GIA_SCRIPT_VIP) {
        alert("❌ Số dư ví không đủ " + formatVND(GIA_SCRIPT_VIP) + " để mua Script này!");
        return;
      }

      const confirmed = confirm("🛒 Xác nhận mua Script này với giá " + formatVND(GIA_SCRIPT_VIP) + "?");
      if (!confirmed) return;

      const buyBtn = document.getElementById("buyBtn");
      const defaultBtnText = "💎 Mua Ngay (Trừ 1.000đ Số Dư Ví)";
      if (buyBtn) {
        buyBtn.innerText = "⏳ Đang xử lý...";
        buyBtn.disabled = true;
      }

      try {
        const res = await fetch(API_URL, {
          method: "POST",
          headers: { "Content-Type": "text/plain;charset=utf-8" },
          body: JSON.stringify({
            action: "deductBalance",
            username: loggedUser,
            postId: POST_ID,
            amount: GIA_SCRIPT_VIP,
            note: "Mua script (ID: " + POST_ID + ")"
          })
        });

        const data = await res.json();

        if (data.status === "success") {
          if (typeof data.balance !== "undefined") {
            syncBalanceUI(data.balance);
          }

          const scriptContent = data.script || "";
          const scriptArea = document.getElementById("scriptText");
          const outputBox = document.getElementById("scriptVipOutput");

          if (scriptArea) scriptArea.value = scriptContent;
          if (outputBox) outputBox.style.display = "block";

          alert("✅ Thanh toán thành công! Giao dịch đã được ghi vào lịch sử.");
        } else {
          if (typeof data.balance !== "undefined") {
            syncBalanceUI(data.balance);
          }
          alert("❌ Lỗi: " + (data.message || "Không thể trừ tiền!"));
        }
      } catch (err) {
        alert("❌ Lỗi kết nối đến máy chủ!");
      } finally {
        if (buyBtn) {
          buyBtn.innerText = defaultBtnText;
          buyBtn.disabled = false;
        }
      }
    };

    window.copyScript = async function () {
      const copyText = document.getElementById("scriptText");
      const copyBtn = document.getElementById("copyBtn");
      if (!copyText || !copyText.value) return;

      const valueToCopy = copyText.value;
      let copied = false;

      try {
        if (navigator.clipboard && window.isSecureContext) {
          await navigator.clipboard.writeText(valueToCopy);
          copied = true;
        }
      } catch (_) {
        copied = false;
      }

      if (!copied) {
        copyText.focus();
        copyText.select();
        copyText.setSelectionRange(0, valueToCopy.length);
        try {
          copied = document.execCommand("copy");
        } catch (_) {
          copied = false;
        }
      }

      if (copied) {
        if (copyBtn) {
          const oldText = copyBtn.innerText;
          copyBtn.innerText = "✅ ĐÃ COPY SCRIPT!";
          setTimeout(() => {
            copyBtn.innerText = oldText;
          }, 1800);
        }
        alert("📋 Đã copy Script thành công!");
      } else {
        alert("⚠️ Không tự động copy được, hãy bôi đen và nhấn Ctrl+C / Cmd+C.");
      }
    };

    window.toggleBaoLoi = function () {
      const box = document.getElementById("formBaoLoiBox");
      if (box) {
        const isHidden = box.style.display === "none" || box.style.display === "";
        box.style.display = isHidden ? "block" : "none";
      }
    };

    window.guiBaoLoi = async function () {
      const loggedUser = localStorage.getItem("logged_user") || "Chưa đăng nhập";
      const tieuDeEl = document.getElementById("loiTieuDe");
      const noiDungEl = document.getElementById("loiNoiDung");
      const sendBtn = document.getElementById("sendReportBtn");

      const tieuDe = tieuDeEl ? tieuDeEl.value : "Khác";
      const noiDungChiTiet = noiDungEl ? noiDungEl.value.trim() : "";

      if (!noiDungChiTiet && tieuDe === "Khác") {
        alert("⚠️ Vui lòng nhập mô tả chi tiết cho mục Khác.");
        return;
      }

      const fullMessage = "[" + tieuDe + "] [ID: " + POST_ID + "]" + (noiDungChiTiet ? " " + noiDungChiTiet : "");

      if (sendBtn) {
        sendBtn.disabled = true;
        sendBtn.innerText = "⏳ Đang gửi...";
      }

      try {
        if (API_URL) {
          await fetch(API_URL, {
            method: "POST",
            headers: { "Content-Type": "text/plain;charset=utf-8" },
            body: JSON.stringify({
              action: "reportIssue",
              postId: POST_ID,
              username: loggedUser,
              message: fullMessage
            })
          });
        }
        alert("✅ Đã gửi báo lỗi tới quản trị viên! Cảm ơn bạn đã phản hồi.");
        if (noiDungEl) noiDungEl.value = "";
        const box = document.getElementById("formBaoLoiBox");
        if (box) box.style.display = "none";
      } catch (_) {
        alert("❌ Không thể gửi báo lỗi lúc này, vui lòng thử lại sau.");
      } finally {
        if (sendBtn) {
          sendBtn.disabled = false;
          sendBtn.innerText = "📤 GỬI BÁO LỖI NGAY";
        }
      }
    };

    setMaintenanceState(false);
  })();
</script>
